async function startMonitoring() { if (tracking) return; if (!window.isSecureContext) { alert("請使用 HTTPS 網址開啟本程式。"); return; } if (!("DeviceMotionEvent" in window)) { alert("此瀏覽器不支援手機動作感測器。"); return; } try { // 第一件事就是直接向手機申請感測器權限 // 不要在這之前執行 await 或音效初始化 if (typeof DeviceMotionEvent.requestPermission === "function") { const permission = await DeviceMotionEvent.requestPermission(); if (permission !== "granted") { alert("感測器權限未開啟，請允許動作與方向感測器。"); return; } } // 重設資料 gravity = null; lastSensorTime = 0; lastPublishTime = 0; highSince = null; lowSince = null; tracking = true; window.addEventListener( "devicemotion", handleMotion, true ); document.getElementById("sensorStatus").textContent = "✅ 地震監測已啟動，等待加速度資料"; publish({ type: "start", active: false, threshold: THRESHOLD, device_id: getDeviceId(), timestamp: new Date().toISOString() }); } catch (error) { console.error("感測器啟動錯誤：", error); alert( "感測器啟動失敗：" + error.name + "：" + error.message ); } }
